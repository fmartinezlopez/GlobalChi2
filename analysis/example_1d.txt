# Example 1D binning
# CCQE-like measurement
#
# Variable indices:
#   0 = Pmu       (muon momentum)
#   1 = Cosmu     (cos theta muon)
#
# Format:
#   block <block_idx> <bin_var_idx>
#   <bin_edge_0> <bin_edge_1> ... <bin_edge_N>

# ═══════════════════════════════════════════════════════════════════
# Block 0: binned in Pmu (var 0)
# ═══════════════════════════════════════════════════════════════════
block 0 0
0.00 0.15 0.30 0.50 0.75 1.20

# ═══════════════════════════════════════════════════════════════════
# Block 1: binned in Cosmu (var 1)
# ═══════════════════════════════════════════════════════════════════
block 1 1
-1.00 -0.50 0.00 0.40 0.70 1.00
